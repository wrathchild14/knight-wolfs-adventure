cmake_minimum_required(VERSION 3.22)
project(KnightWolfsAdventure VERSION 2.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(KWA_BUILD_GAME "Build the graphical game" ON)
include(CTest)
add_library(kwa_core src/core/level.cpp src/core/world.cpp)
target_include_directories(kwa_core PUBLIC include)
if(MSVC)
  target_compile_options(kwa_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(kwa_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
if(BUILD_TESTING)
  add_executable(kwa_tests tests/core_tests.cpp)
  target_link_libraries(kwa_tests PRIVATE kwa_core)
  target_compile_definitions(kwa_tests PRIVATE KWA_LEVEL_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets/levels")
  add_test(NAME gameplay COMMAND kwa_tests)
  add_executable(kwa_journey_tests tests/journey_tests.cpp)
  target_link_libraries(kwa_journey_tests PRIVATE kwa_core)
  target_compile_definitions(kwa_journey_tests PRIVATE KWA_LEVEL_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets/levels")
  add_test(NAME full_journey COMMAND kwa_journey_tests)
  set_tests_properties(full_journey PROPERTIES TIMEOUT 90)
endif()
if(KWA_BUILD_GAME)
  find_package(raylib 5.5 QUIET)
  if(NOT raylib_FOUND)
    include(FetchContent)
    set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(raylib
      URL https://codeload.github.com/raysan5/raylib/tar.gz/refs/tags/5.5
      URL_HASH SHA256=aea98ecf5bc5c5e0b789a76de0083a21a70457050ea4cc2aec7566935f5e258e
      DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(raylib)
    set_property(DIRECTORY "${raylib_SOURCE_DIR}" PROPERTY EXCLUDE_FROM_ALL TRUE)
  endif()
  add_executable(knight_wolf src/main.cpp src/platform/assets.cpp src/platform/renderer.cpp src/platform/application.cpp)
  target_link_libraries(knight_wolf PRIVATE kwa_core raylib)
  target_include_directories(knight_wolf PRIVATE src/platform)
  if(MINGW)
    target_link_options(knight_wolf PRIVATE -static)
  endif()
  if(MSVC)
    target_compile_options(knight_wolf PRIVATE /W4 /permissive-)
  else()
    target_compile_options(knight_wolf PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_custom_command(TARGET knight_wolf POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/Project1/Content" "$<TARGET_FILE_DIR:knight_wolf>/assets/content"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets/levels" "$<TARGET_FILE_DIR:knight_wolf>/assets/levels")
  install(TARGETS knight_wolf RUNTIME DESTINATION .)
  install(DIRECTORY Project1/Content/ DESTINATION assets/content PATTERN "*.mgcb" EXCLUDE PATTERN "*.spritefont" EXCLUDE PATTERN "*.fx" EXCLUDE)
  install(DIRECTORY assets/levels DESTINATION assets)
  install(FILES README.md DESTINATION .)
  install(FILES docs/ARCHITECTURE.md docs/GITHUB_BUILDS.md docs/VALIDATION.md
    docs/THIRD_PARTY.md docs/raylib-LICENSE DESTINATION docs)
  set(CPACK_GENERATOR ZIP)
  include(CPack)
endif()
